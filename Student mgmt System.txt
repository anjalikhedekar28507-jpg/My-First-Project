students = {}
while True:
    print("\n1.Add 2.View 3.Exit")
    ch = input("Enter your choice: ")

    if ch == "1":
        roll = input("Roll No: ")
        name = input("Name: ")
        students[roll] = name

    elif ch == "2":
        for roll, name in students.items():
            print(roll, "-", name)

    elif ch == "3":
        print("Thank You!")
        break

    else:
        print("Inavlid choice")
